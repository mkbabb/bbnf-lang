### AZ.W2.3 Wire-contract test

Create `crates/core/tests/named_type_preservation.rs`:

```rust
fn assert_named_preserved(grammar_path: &str, named_rules: &[(&str, &str)]) {
    let prepared = bbnf::compile_paths_request(
        CompileTarget::Rust,
        &[grammar_path.into()],
        Default::default(),
    ).unwrap();
    for (rule_name, expected_type_name) in named_rules {
        let rule = prepared.ir.rules.iter()
            .find(|r| prepared.ir.get_string(r.name) == *rule_name)
            .unwrap_or_else(|| panic!("rule {rule_name} not found"));
        let expected_sid = prepared.ir.strings.iter()
            .position(|s| s == expected_type_name)
            .map(|p| p as StringId)
            .unwrap_or_else(|| panic!("type name {expected_type_name} not interned"));
        let actual = prepared.ir.types.iter()
            .find(|(id, _)| *id == rule.id)
            .map(|(_, td)| td);
        assert_eq!(actual, Some(&TypeDesc::Named(expected_sid)));
    }
}

#[test]
fn css_l4_named_types() {
    assert_named_preserved(
        "../../grammar/css/l4/stylesheet.bbnf",
        &[("colorFunction", "Color"), ("colorFn", "Color"), ("colorMix", "Color")],
    );
}

#[test]
fn json_named_types() {
    assert_named_preserved(
        "../../grammar/json/json.bbnf",
        &[("string", "String")],
    );
}
```
