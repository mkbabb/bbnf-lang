fn assert_cssparser_parity(fixture: &str) {
    let src = load_fixture(fixture);
    let bbnf_canonical = CssL4Parser::stylesheet_prettify().parse(&src).unwrap();
    let cssparser_tokens = cssparser::Parser::new(&mut cssparser::ParserInput::new(&src));

    assert_eq!(token_normalize(&bbnf_canonical), token_normalize(&cssparser_canonical));
}
